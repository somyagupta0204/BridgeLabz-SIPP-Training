import java.util.*;

class User {
    protected String username;
    protected String password;

    public User(String username, String password) {
        this.username = username;
        this.password = password;
    }

    public boolean login() {
        System.out.println("User logged in with username: " + username);
        return true;
    }
}

class Student extends User {
    private boolean enrolled;

    public Student(String username, String password, boolean enrolled) {
        super(username, password);
        this.enrolled = enrolled;
    }

    @Override
    public boolean login() {
        if (enrolled) {
            System.out.println("Student " + username + " logged in. Access to exams granted.");
            return true;
        } else {
            System.out.println("Student " + username + " login failed. Not enrolled in any course.");
            return false;
        }
    }
}

class Admin extends User {
    private boolean isAdmin;

    public Admin(String username, String password, boolean isAdmin) {
        super(username, password);
        this.isAdmin = isAdmin;
    }

    @Override
    public boolean login() {
        if (isAdmin) {
            System.out.println("Admin " + username + " logged in. Exam management access granted.");
            return true;
        } else {
            System.out.println("Admin " + username + " login failed. Insufficient privileges.");
            return false;
        }
    }
}

public class OnlineExamPortal {
    public static void main(String[] args) {
        User s1 = new Student("john123", "pass123", true);
        User s2 = new Student("alice456", "pass456", false);
        User a1 = new Admin("admin01", "adminpass", true);
        User a2 = new Admin("staff01", "staffpass", false);

        s1.login();
        s2.login();
        a1.login();
        a2.login();
    }
}
