ques-2 Attendance Marker

import java.util.*;

public class AttendanceMarker {
    public static void main(String[] args) {
        int numStudents = 3;
        int numDays = 30;
        char[][] attendance = new char[numStudents][numDays];

        // Mark random attendance for demo (P = Present, A = Absent)
        Random rand = new Random();
        for (int i = 0; i < numStudents; i++) {
            for (int j = 0; j < numDays; j++) {
                attendance[i][j] = rand.nextBoolean() ? 'P' : 'A';
            }
        }

        // Update attendance: Student 1, Day 3 -> P
        attendance[0][2] = 'P';

        // Generate monthly report
        for (int i = 0; i < numStudents; i++) {
            int presentCount = 0;
            for (int j = 0; j < numDays; j++) {
                if (attendance[i][j] == 'P') presentCount++;
            }
            System.out.println("Student " + (i+1) + " was present for " + presentCount + " days.");
        }

        // Optional: Print full attendance table
        System.out.println("\nAttendance Table:");
        for (int i = 0; i < numStudents; i++) {
            System.out.print("Student " + (i+1) + ": ");
            for (int j = 0; j < numDays; j++) {
                System.out.print(attendance[i][j] + " ");
            }
            System.out.println();
        }
    }
}
