ques-3 Student Score Storage

import java.util.*;
import java.util.stream.*;

public class StudentScores {
    public static void main(String[] args) {
        Map<String, Integer> scores = new HashMap<>();
        scores.put("Alice", 85);
        scores.put("Bob", 92);
        scores.put("Charlie", 78);
        scores.put("David", 95);
        scores.put("Eva", 88);

        Map.Entry<String, Integer> topScorer = scores.entrySet()
                .stream()
                .max(Map.Entry.comparingByValue())
                .get();

        System.out.println("Top Scorer: " + topScorer.getKey() + " with " + topScorer.getValue() + " marks");
    }
}
