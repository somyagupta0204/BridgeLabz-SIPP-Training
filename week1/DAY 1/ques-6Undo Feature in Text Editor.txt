ques-2Undo Feature in Text Editor

import java.util.Stack;

class TextEditor {
    String text = "";
    Stack<String> undoStack = new Stack<>();

    void type(String newText) {
        undoStack.push(text);
        text = text + newText;
    }

    void undo() {
        if (!undoStack.isEmpty()) {
            text = undoStack.pop();
        } else {
            System.out.println("Nothing to undo!");
        }
    }

    void showText() {
        System.out.println("Current Text: " + text);
    }
}

public class UndoFeature {
    public static void main(String[] args) {
        TextEditor editor = new TextEditor();

        editor.type("Hello ");
        editor.showText();

        editor.type("World!");
        editor.showText();

        editor.undo();
        editor.showText();

        editor.undo();
        editor.showText();
    }
}
